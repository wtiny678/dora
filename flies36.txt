#RANK, DETERMINANT AND TRACE OF AN ARRAY.
import numpy as np
a=np.array([[4,5,6],[7,1,4],[5,5,5]])
rank=np.linalg.matrix_rank(a)
tra=a.trace()
deter=np.linalg.det(a)
print('Matrix=',a)
print('Determinant of Matrix:',deter)
print('Rank of Matrix',rank)
print('Trace of Matrix',tra)

#(DOT, INNER, OUTER, PRODUCT), MATRIX EXPONENTIATION
import numpy as np
from numpy.linalg import matrix_power
a = np.array([2, 6])
b = np.array([3, 10])
print("Vectors :")
print("a = ", a)
print("\nb = ", b)
print("\nInner product of vectors a and b =")
print(np.inner(a, b))
print("\nOuter Product of Vectors a and b=")
print(np.outer(a,b))
print("\nDot Product of Vectors a and b=")
print(np.dot(a,b))
print("\nDot Product of Vector a=")
print(np.prod(a))
x = np.array([[2, 3, 4], [3, 2, 9],[3,4,5]])
y = np.array([[1, 5, 0], [5, 10, 3],[3,2,1]])
print("\nMatrices :")
print("x =", x)
print("\ny =", y)
print("\nInner product of matrices x and y =")
print(np.inner(x, y))
print("\nOuter Product of matrices x and y=")
print(np.outer(x,y))
print("\nDot Product of matrices x and y=")
print(np.dot(x,y))
print("\nDot Product of matrice a=")
print(np.prod(x))
print("\nExponentiation of matrice a=")
print(matrix_power(x, 4))


#LINEAR ALGEBRA 
import numpy as np
arr1 = np.array([[1, 2], [3, 5]])
arr2 = np.array([1, 2])
print("Array1",arr1)
print("Array2",arr2)

#SORTING, SEARCHING, AND COUNTING
import numpy as np
a = np.array([10, 5, 20, 5, 30, 15])
print("Original Array:")
print(a)
print("Sorting:")
print(np.sort(a))
print("Reverse Sorting:")
print(np.sort(a)[::-1])
print("Searching:")
print(np.where(a == 5))
print("Counting:")
print(np.count_nonzero(a == 5))
