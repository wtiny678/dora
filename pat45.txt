#DataFrame and Series into numpy.array

import pandas as pd
import numpy as np
df = pd.DataFrame({
    'A': [1, 2, 3],
    'B': [4, 5, 6]
})

print("DataFrame:")
print(df)

arr1 = df.to_numpy()
print("DataFrame to NumPy Array:")
print(arr1)
s = pd.Series([10, 20, 30])
print("Series:")
print(s)
arr2 = s.to_numpy()
print("Series to NumPy Array:")
print(arr2)
