#MERGE 
import pandas as pd 
data1 = pd.read_csv("D:/Data.csv")
data2 = pd.read_csv("D:/csv/data2.csv")
df = pd.DataFrame(data1,index=[0, 1, 2, 3])
df1 = pd.DataFrame(data2, index=[2, 5, 6, 7])
frames = [df, df1]
res1 = pd.concat(frames)
print(res1)

print("Inner Join")
res2 = pd.concat([df, df1], axis=1, join='inner')
print(res2)

print("Outer Join")
res2 = pd.concat([df, df1], axis=1, join='outer')
print(res2)

print("Merging")
res = pd.merge(df, df1, on='Sno')
print(res)

print("Left Merging")
res = pd.merge(df, df1, how='left', on=['Sno','Rollno'])
print(res)

print("Right Merging")
res = pd.merge(df, df1, how='right', on=['Sno','Rollno'])
print(res)

print("Outer Merging")
res = pd.merge(df, df1, how='outer', on=['Sno','Rollno'])
print(res)

print("Inner Merging")
res = pd.merge(df, df1, how='inner', on=['Sno','Rollno'])
print(res)
